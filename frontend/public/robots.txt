# Per robots.txt's actual matching rule (RFC 9309): a crawler that
# matches a specific User-agent group uses ONLY that group's directives
# — it does NOT also inherit the wildcard (*) group's rules. So every
# named group below repeats the Disallow lines explicitly; leaving them
# out would have silently allowed these crawlers into /app and /auth.
User-agent: *
Allow: /
Disallow: /app
Disallow: /auth
